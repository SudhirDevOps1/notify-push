💻 Copy-Paste HTML Code into Your Website

Add this single snippet to any webpage to send notifications: 

<!-- 1. Include NotifyPush Client -->
<script src="https://cdn.jsdelivr.net/gh/SudhirDevOps1/notify-push@main/web/notifypush.js"></script>

<!-- 2. Send an Alert -->
<script>
  NotifyPush.send({
    topic: 'desktop-alerts-5252',
    title: 'Website Alert 🔔',
    message: 'User completed checkout!',
    priority: 'high',
    tags: ['moneybag', 'bell']
  });
</script>
